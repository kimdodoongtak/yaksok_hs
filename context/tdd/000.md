### DB 검증 재현 명령

Node.js의 TypeScript 실행에서 확장자 없는 상대 import를 해석하는 hook을 검증 프로세스에만 등록했다. `react-server` 조건으로 `server-only`를 유지한 실제 앱 모듈을 불러온다.

```bash
docker compose exec -T tools node --conditions=react-server --input-type=module <<'JS'
import assert from 'node:assert/strict';
import { registerHooks } from 'node:module';
registerHooks({
  resolve(specifier, context, nextResolve) {
    try { return nextResolve(specifier, context); }
    catch (error) {
      if (error.code === 'ERR_MODULE_NOT_FOUND' && specifier.startsWith('.')) {
        return nextResolve(`${specifier}.ts`, context);
      }
      throw error;
    }
  }
});
const target = new URL(process.env.DATABASE_URL);
assert.equal(target.hostname, 'db');
assert.equal(target.pathname, '/sdd');
const { prisma } = await import('./src/lib/prisma.ts');
try {
  const result = await prisma.$queryRaw`SELECT 1 AS value`;
  assert.deepEqual(result, [{ value: 1 }]);
  const reloaded = await import('./src/lib/prisma.ts?reload=1');
  assert.equal(reloaded.prisma, prisma);
  console.log('PASS: local PostgreSQL SELECT 1; Prisma module reload reuses client');
} finally {
  await prisma.$disconnect();
}
JS
```
